import puppeteer from 'puppeteer';
import fs from 'fs';
import { Parser } from 'json2csv';

async function rasparDados() {
  console.log('Iniciando o navegador...');
  const browser = await puppeteer.launch({
    headless: true,
    slowMo: 50,
    defaultViewport: null,
    args: ['--start-maximized'],
  });

  const page = await browser.newPage();
  console.log('Nova página aberta.');

  // --- TÉCNICAS AVANÇADAS DE EVASÃO ---
  await page.setUserAgent('Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/108.0.0.0 Safari/537.36');
  await page.setViewport({ width: 1920, height: 1080 });
  await page.evaluateOnNewDocument(() => { Object.defineProperty(navigator, 'webdriver', { get: () => false }) });

  // --- LÓGICA DE LOGIN ---
  console.log('Navegando para a página de login...');
  await page.goto('https://unoerpwiki.herokuapp.com/login', { waitUntil: 'networkidle2' });
  await page.type('#input-20', 'unolab@unosolucoes.com.br');
  await page.type('#input-22', 'unolab@2025');
  await new Promise(resolve => setTimeout(resolve, 1000));
  await page.click('button.mt-2.text-none.v-btn.v-btn--contained.theme--dark.v-size--large.blue.darken-2');
  await page.waitForNavigation({ waitUntil: 'networkidle2' });
  console.log('Login realizado com sucesso!');
  
  const supportPageUrl = 'https://unoerpwiki.herokuapp.com/Suporte';
  await page.goto(supportPageUrl, { waitUntil: 'networkidle2' });
  console.log('Página de suporte carregada.');

  const allResults = [];
  const numCategorias = await page.$$eval('blockquote', (blocks) => blocks.length);
  console.log(`Encontradas ${numCategorias} categorias para processar.`);

  for (let i = 1; i <= numCategorias; i++) {
    const seletorCategoriaAtual = `blockquote:nth-of-type(${i}) > h1.toc-header`;
    let textoCategoria = await page.$eval(seletorCategoriaAtual, (el) => el.innerText.replace('¶', '').trim());
    console.log(`\n--- Processando Categoria: ${textoCategoria} ---`);

    // LÓGICA DE SUBCATEGORIAS PARA "Vendas"
    if (textoCategoria.includes('Vendas')) {
        console.log('Categoria "Vendas" detectada, aplicando lógica de subcategorias...');
        const elementsBetween = await page.evaluate((selector) => {
            const startNode = document.querySelector(selector);
            let nextNode = startNode.parentElement.nextElementSibling;
            const content = [];
            while (nextNode && nextNode.tagName !== 'BLOCKQUOTE') {
                if (nextNode.tagName === 'P' && nextNode.querySelector('span.text-huge')) {
                    content.push({ type: 'subcategory', text: nextNode.innerText.trim() });
                } else if (nextNode.tagName === 'UL') {
                    const links = Array.from(nextNode.querySelectorAll('li > a.is-internal-link, li > a.is-system-link'));
                    // Para evitar o erro de "stale element", passamos o HTML do link
                    content.push({ type: 'list', links: links.map(a => a.outerHTML) });
                }
                nextNode = nextNode.nextElementSibling;
            }
            return content;
        }, seletorCategoriaAtual);

        let currentSubcategory = '';
        for (const element of elementsBetween) {
            if (element.type === 'subcategory') {
                currentSubcategory = element.text;
                console.log(`  > Subcategoria encontrada: ${currentSubcategory}`);
            } else if (element.type === 'list') {
                for (const linkHTML of element.links) {
                    // Recria o elemento temporariamente para poder interagir com ele
                    const tempHandle = await page.evaluateHandle(html => {
                        const div = document.createElement('div');
                        div.innerHTML = html;
                        return div.firstElementChild;
                    }, linkHTML);
                    
                    const textoTitulo = await tempHandle.evaluate(a => a.innerText.trim());
                    console.log(`    > Processando: ${textoTitulo}`);

                    await tempHandle.click();
                    await new Promise(resolve => setTimeout(resolve, 1500));
                    
                    const finalCategoryName = `${textoCategoria} - ${currentSubcategory}`;

                    const textoConteudo = await page.evaluate(() => {
                        const contentElement = document.querySelector('div.contents');
                        if (!contentElement) return 'Conteúdo não encontrado.';
                        return contentElement.innerText.split('\n').filter(line => line.trim() !== '').join('\n');
                    });
            
                    let palavrasChave = 'Nenhuma palavra-chave encontrada';
                    try {
                        const seletorPalavraChave = 'a.v-chip span.teal--text.text--darken-2';
                        await page.waitForSelector(seletorPalavraChave, { timeout: 3000 });
                        palavrasChave = await page.evaluate((seletor) => {
                            const spans = Array.from(document.querySelectorAll(seletor));
                            return spans.map(span => span.innerText.trim()).filter(text => text.length > 0).join(', ');
                        }, seletorPalavraChave);
                    } catch (error) { /* Normal */ }
            
                    allResults.push({ 'Categoria': finalCategoryName, 'Titulo': textoTitulo, 'Conteudo': textoConteudo, 'Palavras Chave': palavrasChave });
                    console.log(`      -> Sucesso! Total de artigos no array: ${allResults.length}`);
            
                    await page.goto(supportPageUrl, { waitUntil: 'networkidle2' });
                }
            }
        }
    } else { // --- LÓGICA PADRÃO PARA OUTRAS CATEGORIAS ---
        const seletorListaAtual = `blockquote:nth-of-type(${i}) + div + ul`;
        const numLinksNaCategoria = await page.$eval(seletorListaAtual, (ul) => 
            ul.querySelectorAll('li > a.is-internal-link, li > a.is-system-link').length
        ).catch(() => 0);
        
        console.log(`Encontrados ${numLinksNaCategoria} links nesta categoria.`);

        for (let j = 1; j <= numLinksNaCategoria; j++) {
            const seletorBaseLi = `${seletorListaAtual} > li:nth-child(${j})`;
            const seletorLinkAtual = `${seletorBaseLi} > a.is-internal-link, ${seletorBaseLi} > a.is-system-link`;
            
            try {
                await page.waitForSelector(seletorLinkAtual, { timeout: 1000 });
                let textoTitulo = await page.$eval(seletorLinkAtual, (a) => a.innerText.trim());

                if (textoCategoria.includes('Processos fora da aplicação') && textoTitulo === 'Mapa mental - case de testes') {
                    console.log(`  > IGNORANDO [${j}/${numLinksNaCategoria}]: ${textoTitulo} conforme regra.`);
                    continue;
                }

                if (textoCategoria.includes('Outros')) {
                    const textoExtra = await page.$eval(`${seletorBaseLi} > mark.pen-red`, (mark) => mark.innerText.trim()).catch(() => '');
                    if (textoExtra) {
                        textoTitulo += textoExtra;
                    }
                }

                console.log(`  > Processando [${j}/${numLinksNaCategoria}]: ${textoTitulo}`);
                await page.click(seletorLinkAtual);
                await new Promise(resolve => setTimeout(resolve, 1500));
                
                const textoConteudo = await page.evaluate(() => {
                  const contentElement = document.querySelector('div.contents');
                  if (!contentElement) return 'Conteúdo não encontrado.';
                  return contentElement.innerText.split('\n').filter(line => line.trim() !== '').join('\n');
                });
                let palavrasChave = 'Nenhuma palavra-chave encontrada';
                try {
                  const seletorPalavraChave = 'a.v-chip span.teal--text.text--darken-2';
                  await page.waitForSelector(seletorPalavraChave, { timeout: 3000 });
                  palavrasChave = await page.evaluate((seletor) => {
                    const spans = Array.from(document.querySelectorAll(seletor));
                    return spans.map(span => span.innerText.trim()).filter(text => text.length > 0).join(', ');
                  }, seletorPalavraChave);
                } catch (error) { /* Normal */ }
                
                allResults.push({ 'Categoria': textoCategoria, 'Titulo': textoTitulo, 'Conteudo': textoConteudo, 'Palavras Chave': palavrasChave });
                console.log(`    -> Sucesso! Total de artigos no array: ${allResults.length}`);

                await page.goto(supportPageUrl, { waitUntil: 'networkidle2' });
            } catch(e) {
                console.log(`  !! Link ${j} na categoria "${textoCategoria}" não encontrado ou falhou, pulando: ${e.message}`);
                continue;
            }
        }
    }

    // REGRA DE TÉRMINO: Após processar todos os links de uma categoria, verifica se era "Clientes".
    if (textoCategoria.includes('Clientes')) {
        console.log("\n--- Última categoria 'Clientes' processada. Encerrando a extração. ---");
        break; // Interrompe o loop principal de categorias, impedindo de ir para a próxima.
    }
  }

  // --- GERAR PLANILHA NO FINAL DE TUDO ---
  if (allResults.length > 0) {
    console.log(`\nExtração finalizada. Foram coletados ${allResults.length} artigos.`);
    const parser = new Parser();
    const csv = parser.parse(allResults);
    fs.writeFileSync('extracao_completa_final.csv', '\ufeff' + csv, 'utf-8');
    console.log('Arquivo "extracao_completa_final.csv" salvo com sucesso!');
  } else {
    console.log('Nenhum dado foi coletado para salvar.');
  }

  await browser.close();
  console.log('Processo finalizado.');
}

rasparDados();
