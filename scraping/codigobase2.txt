import puppeteer from 'puppeteer';
import fs from 'fs';
import { Parser } from 'json2csv';

async function rasparDados() {
  console.log('Iniciando o navegador...');
  const browser = await puppeteer.launch({
    headless: true, // Pode manter true
    slowMo: 50,
    defaultViewport: null,
    args: ['--start-maximized'],
  });

  const page = await browser.newPage();
  console.log('Nova página aberta.');

  // --- TÉCNICAS AVANÇADAS DE EVASÃO ---
  await page.setUserAgent('Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/108.0.0.0 Safari/537.36');
  await page.setViewport({ width: 1920, height: 1080 });
  await page.evaluateOnNewDocument(() => { Object.defineProperty(navigator, 'webdriver', { get: () => false }) });

  // --- LÓGICA DE LOGIN ---
  console.log('Navegando para a página de login...');
  await page.goto('https://unoerpwiki.herokuapp.com/login', { waitUntil: 'networkidle2' });
  await page.type('#input-20', 'unolab@unosolucoes.com.br');
  await page.type('#input-22', 'unolab@2025');
  await new Promise(resolve => setTimeout(resolve, 1000));
  await page.click('button.mt-2.text-none.v-btn.v-btn--contained.theme--dark.v-size--large.blue.darken-2');
  
  // Mantém networkidle2 aqui para garantir que o login foi bem sucedido
  await page.waitForNavigation({ waitUntil: 'networkidle2' }); 
  console.log('Login realizado com sucesso!');
  
  // --- ETAPA 1: MAPEAMENTO USANDO A LÓGICA QUE FUNCIONOU ---
  const supportPageUrl = 'https://unoerpwiki.herokuapp.com/Suporte';
  await page.goto(supportPageUrl);
  await new Promise(resolve => setTimeout(resolve, 1500)); // Espera com Promise
  console.log('Página de suporte carregada. Criando mapa de extração...');

  const articlesToScrape = await page.evaluate(() => {
    const articles = [];
    // Encontra todos os cabeçalhos de categoria
    const headers = document.querySelectorAll('blockquote > h1.toc-header');

    for (const header of headers) {
      const categoryName = header.innerText.replace('¶', '').trim();
      
      // A partir do <blockquote> do header, procura o próximo irmão <ul>
      let nextElement = header.parentElement.nextElementSibling;
      while (nextElement) {
        if (nextElement.tagName === 'UL') {
          // Encontrou a lista, agora extrai os dados dos links
          const links = nextElement.querySelectorAll('li > a.is-internal-link, li > a.is-system-link');
          for (const link of links) {
            articles.push({
              category: categoryName,
              title: link.innerText.trim(),
              href: link.getAttribute('href')
            });
          }
          break; // Para de procurar depois de encontrar a primeira lista
        }
        if (nextElement.tagName === 'BLOCKQUOTE') {
            break; // Se encontrar outra categoria, para
        }
        nextElement = nextElement.nextElementSibling;
      }
    }
    return articles;
  });

  console.log('--- Rotas extraídas ---');
  articlesToScrape.forEach(article => console.log(`- ${article.href}`));
  
  const totalArticles = articlesToScrape.length;
  console.log(`\nMapeamento concluído. ${totalArticles} artigos encontrados.`);
  
  const allResults = [];

  // --- ETAPA 2: LOOP E EXTRAÇÃO (APENAS DA PRIMEIRA CATEGORIA) ---
  const primeiraCategoriaNome = articlesToScrape[0]?.category;
  if (!primeiraCategoriaNome) {
    console.log('Nenhum artigo encontrado.');
  } else {
    console.log(`\n--- Processando apenas a primeira categoria: ${primeiraCategoriaNome} ---`);
    const linksDaPrimeiraCategoria = articlesToScrape.filter(a => a.category === primeiraCategoriaNome);
    
    for (const article of linksDaPrimeiraCategoria) {
      if (!article.href) continue;
      
      const url = `https://unoerpwiki.herokuapp.com${article.href}`;
      console.log(`  > Processando [${allResults.length + 1}/${linksDaPrimeiraCategoria.length}]: ${article.title}`);

      try {
        await page.goto(url);
        await new Promise(resolve => setTimeout(resolve, 1500)); // Espera com Promise

        const textoConteudo = await page.evaluate(() => {
          const el = document.querySelector('div.contents');
          return el ? el.innerText.split('\n').filter(l => l.trim() !== '').join('\n') : '';
        });
        
        let palavrasChave = 'Nenhuma palavra-chave encontrada';
        try {
          const sel = 'a.v-chip span.teal--text.text--darken-2';
          await page.waitForSelector(sel, { timeout: 3000 });
          palavrasChave = await page.evaluate((s) => Array.from(document.querySelectorAll(s)).map(span => span.innerText.trim()).filter(t => t.length > 0).join(', '), sel);
        } catch (error) {}

        allResults.push({
          'Categoria': article.category, 'Titulo': article.title, 'Conteudo': textoConteudo, 'Palavras Chave': palavrasChave, 'URL': url
        });
      } catch (e) {
        console.error(`  !! Erro ao processar o link "${article.title}": ${e.message}`);
      }
    }
  }

  // --- ETAPA 3: GERAR PLANILHA ---
  if (allResults.length > 0) {
    console.log(`\nExtração da primeira categoria finalizada. Foram coletados ${allResults.length} artigos.`);
    const parser = new Parser();
    const csv = parser.parse(allResults);
    fs.writeFileSync('extracao_primeira_categoria.csv', '\ufeff' + csv, 'utf-8');
    console.log('Arquivo "extracao_primeira_categoria.csv" salvo com sucesso!');
  } else {
    console.log('\nNenhum dado foi extraído.');
  }

  await browser.close();
  console.log('Processo finalizado.');
}

rasparDados();
